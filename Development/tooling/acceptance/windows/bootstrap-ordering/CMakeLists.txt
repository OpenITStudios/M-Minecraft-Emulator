cmake_minimum_required(VERSION 3.24)
project(BootstrapOrderingSelfTest LANGUAGES CXX)

option(BOOTSTRAP_ORDERING_TEST_ONLY "Build the standalone self-child ordering experiment" OFF)
option(BOOTSTRAP_GRAPHICS_WARM_POOLS_EXPERIMENT "Enable version-bound DefaultBufferHeap startup-capacity experiment" OFF)
option(BOOTSTRAP_UPLOAD_RETENTION_EXPERIMENT "Enable version-bound Upload empty-heap retention canary" OFF)
if(BOOTSTRAP_GRAPHICS_WARM_POOLS_EXPERIMENT AND BOOTSTRAP_UPLOAD_RETENTION_EXPERIMENT)
  message(FATAL_ERROR "Heap experiments must be evaluated independently")
endif()
if(NOT BOOTSTRAP_ORDERING_TEST_ONLY)
  message(FATAL_ERROR "Explicit BOOTSTRAP_ORDERING_TEST_ONLY=ON is required")
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This standalone probe requires Windows MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DETOURS_ROOT "" CACHE PATH "Existing trusted Microsoft Detours checkout; never downloaded")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${DETOURS_ROOT}" rev-parse HEAD
  OUTPUT_VARIABLE detours_head OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE head_result)
if(NOT head_result EQUAL 0 OR NOT detours_head STREQUAL "1dbd4ea281d73d478aa8d09a8b83fbd265f715f7")
  message(FATAL_ERROR "DETOURS_ROOT must be the reviewed Detours commit")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${DETOURS_ROOT}" diff --quiet HEAD -- src
  RESULT_VARIABLE dirty_result)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${DETOURS_ROOT}" ls-files --others --exclude-standard src
  OUTPUT_VARIABLE untracked RESULT_VARIABLE untracked_result)
if(NOT dirty_result EQUAL 0 OR NOT untracked_result EQUAL 0 OR NOT untracked STREQUAL "")
  message(FATAL_ERROR "Detours src must be clean and have no untracked source")
endif()

# Separate test-only library; no vendoring, download, prebuilt DLL or product linkage.
set(detours_sources detours modules disasm image creatwth
  disolx86 disolx64 disolia64 disolarm disolarm64)
foreach(source IN LISTS detours_sources)
  list(APPEND detours_files "${DETOURS_ROOT}/src/${source}.cpp")
endforeach()
add_library(probe-detours STATIC ${detours_files})
target_include_directories(probe-detours PUBLIC "${DETOURS_ROOT}/src")
target_compile_definitions(probe-detours PRIVATE DETOUR_DEBUG=0 WIN32_LEAN_AND_MEAN)
target_compile_options(probe-detours PRIVATE /W4 /WX)

add_library(bootstrap-dependency SHARED dependency.cpp)
add_library(bootstrap-compat SHARED compat.cpp compat.def presentation-enrollment.cpp dxgi-pacing.cpp fps-hud.cpp graphics-warm-pools.cpp)
if(BOOTSTRAP_GRAPHICS_WARM_POOLS_EXPERIMENT)
  target_compile_definitions(bootstrap-compat PRIVATE BOOTSTRAP_GRAPHICS_WARM_POOLS_EXPERIMENT=1)
  target_link_options(bootstrap-compat PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/bootstrap-compat.map")
endif()
if(BOOTSTRAP_UPLOAD_RETENTION_EXPERIMENT)
  target_compile_definitions(bootstrap-compat PRIVATE BOOTSTRAP_UPLOAD_RETENTION_EXPERIMENT=1)
  target_link_options(bootstrap-compat PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/bootstrap-compat.map")
endif()
target_link_libraries(bootstrap-compat PRIVATE probe-detours shell32 ole32 uuid advapi32 dxgi user32 gdi32 bcrypt)
add_library(bootstrap-enroll SHARED enroll.cpp)
target_link_libraries(bootstrap-enroll PRIVATE probe-detours advapi32 dxgi)
add_executable(bootstrap-ordering probe.cpp)
target_link_libraries(bootstrap-ordering PRIVATE bootstrap-dependency probe-detours shell32 ole32 oleaut32 uuid advapi32 d3d12 dxgi user32)
add_dependencies(bootstrap-ordering bootstrap-compat bootstrap-enroll)
add_executable(frame-pacing-control frame-pacing-control.cpp dxgi-pacing.cpp fps-hud.cpp)
target_link_libraries(frame-pacing-control PRIVATE d3d11 d3d12 dxgi user32 gdi32)
add_executable(fps-hud-control fps-hud-control.cpp fps-hud.cpp)
add_executable(hud-lifetime-control hud-lifetime-control.cpp fps-hud.cpp)
target_link_libraries(hud-lifetime-control PRIVATE user32 gdi32)
target_link_libraries(fps-hud-control PRIVATE user32 gdi32)
add_executable(graphics-warm-pools-control graphics-warm-pools-control.cpp graphics-warm-pools.cpp)
target_link_libraries(graphics-warm-pools-control PRIVATE probe-detours bcrypt)
if(BOOTSTRAP_UPLOAD_RETENTION_EXPERIMENT)
  target_compile_definitions(graphics-warm-pools-control PRIVATE BOOTSTRAP_UPLOAD_RETENTION_EXPERIMENT=1)
else()
  target_compile_definitions(graphics-warm-pools-control PRIVATE BOOTSTRAP_GRAPHICS_WARM_POOLS_EXPERIMENT=1)
endif()
add_executable(education-slot-control education-slot-control.cpp)
foreach(target education-slot-control bootstrap-ordering bootstrap-dependency bootstrap-compat bootstrap-enroll frame-pacing-control fps-hud-control hud-lifetime-control graphics-warm-pools-control)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
    _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
endforeach()
# Deliberately no install, CTest registration, product dependency or automatic execution.
