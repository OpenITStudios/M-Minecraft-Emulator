    static SURFACE: Mutex<Surfaces> = Mutex::new(Surfaces {active:None,parked:Vec::new(),suspended:false});
    fn same(lease:&Lease,target:&OwnedProcess)->bool {
        lease.saved.as_ref().is_some_and(|s| s.target.pid==target.pid && s.target.created_file_time==target.created_file_time && s.target.session_id==target.session_id && s.target.window_handle==target.window_handle)
    }
    pub fn is_active() -> bool {
        SURFACE.try_lock().map(|s| s.active.is_some() || !s.parked.is_empty()).unwrap_or(true)
    }
    pub fn release() -> io::Result<()> {
        let mut state=SURFACE.lock().map_err(|_|io::Error::other("Workspace state unavailable"))?;
        let mut error=None;
        if let Some(active)=state.active.as_mut(){match active.stop(){Ok(())=>state.active=None,Err(e)=>error=Some(e)}}
        state.parked.retain_mut(|lease| match lease.stop(){Ok(())=>false,Err(e)=>{if error.is_none(){error=Some(e);}true}});
        if let Some(error)=error{return Err(error);}
        state.suspended=false;Ok(())
    }
    pub fn suspend()->io::Result<()> {
        let mut state=SURFACE.lock().map_err(|_|io::Error::other("Workspace state unavailable"))?;
        if let Some(lease)=state.active.as_mut(){if !matches!(lease.send(Command::Hide)?,Reply::Hidden){return Err(io::Error::other("Surface suspend failed"));}}
        state.suspended=true;Ok(())
    }
    pub fn release_target(target:&OwnedProcess)->io::Result<()> {
        let mut state=SURFACE.lock().map_err(|_|io::Error::other("Workspace state unavailable"))?;
        if state.active.as_ref().is_some_and(|lease|same(lease,target)){
            state.active.as_mut().unwrap().stop()?;state.active=None;return Ok(());
        }
        if let Some(index)=state.parked.iter().position(|lease|same(lease,target)){
            state.parked[index].stop()?;state.parked.remove(index);return Ok(());
        }
        Err(io::Error::other("Workspace target no longer managed"))
    }
